const Timer = ({ startTime }: { startTime: number }) => {
  const [now, setNow] = useState(Date.now());

  useEffect(() => {
    const interval = setInterval(() => {
      setNow(Date.now());
    }, 100); // 使用更短的间隔确保秒数切换精准，不跳秒
    return () => clearInterval(interval);
  }, []);

  const formatTime = (currentNow: number) => {
    const elapsedMs = Math.max(0, currentNow - startTime);
    
    const totalSeconds = Math.floor(elapsedMs / 1000);
    const h = Math.floor(totalSeconds / 3600).toString().padStart(2, '0');
    const m = Math.floor((totalSeconds % 3600) / 60).toString().padStart(2, '0');
    const s = (totalSeconds % 60).toString().padStart(2, '0');
    return `${h}:${m}:${s}`;
  };

  return (
    <div className="flex items-center gap-2 bg-black/40 backdrop-blur-md px-4 py-2 rounded-full border border-white/20">
      <Clock className="w-4 h-4 text-white" />
      <motion.span 
        animate={{ opacity: [0.7, 1, 0.7] }}
        transition={{ repeat: Infinity, duration: 2, ease: "easeInOut" }}
        className="font-mono text-xl font-bold tracking-wider bg-clip-text text-transparent bg-gradient-to-r from-red-500 to-yellow-500"
      >
        {formatTime(now)}
      </motion.span>
    </div>
  );
};
